void ColorTsdfVolume::getBoundingBox(OutputArray bb, int precision) const { if (precision == Volume::BoundingBoxPrecision::VOXEL) { CV_Error(Error::StsNotImplemented, "Voxel mode is not implemented yet"); } else { float sz = this->settings.getVoxelSize(); Vec3f res; this->settings.getVolumeResolution(res); Vec3f volSize = res * sz; Vec6f(0, 0, 0, volSize[0], volSize[1], volSize[2]).copyTo(bb); } }
