void GuiReceiver::createWindow(QString name, int flags)
{
    if (!qApp)
        CV_Error(cv::Error::StsNullPtr, "NULL session handler" );

    // Check the name in the storage
    if (icvFindWindowByName(name))
    {
        return;
    }

    nb_windows++;
    new CvWindow(name, flags);
    waitKeyImpl(1);
}
