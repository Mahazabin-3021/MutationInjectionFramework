inline size_t total(const MatShape& shape, int start = -1, int end = -1)
{
    //if (shape.empty())
    //    return 0;

    int dims = (int)shape.size();

    if (start == -1) start = 0;
    if (end == -1) end = dims;

    CV_CheckLE(0, start, "");
    CV_CheckLE(start, end, "");
    CV_CheckLE(end, dims, "");

    size_t elems = 1;
    for (int i = start; i < end; i++)
    {
        elems *= (size_t)shape[i];
    }
    return elems;
}
